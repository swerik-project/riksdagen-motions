OCR Quality Estimate

Summary

We estimate OCR quality in a sample.

  • 3 lines from three pages per year and chamber
  • word-error rate (WER)
  • character-error rate (CER)
  • levenshtein distance (LEV)

What is the problem

We want to understand how closely OCR-rendered text matches the original.

Estimation procedure

We calculate WER, CER and LEV between manually annotated lines (directly from PDF) and the identified most-likely line (XML).

Sampling plan [if applicable]

3 lines are annotated from three randomly selected pages per year, per chamber.

Annotation guidelines [if applicable]

Annotator first counts the lines and adds the number of lines to the .csv annotation file. A script then selects a three random, non-repeating numbers from the range 1--number of lines on page. These are the three lines to annotate.

Annotator writes down exactly what is contained in the specified line in the csv file.

102    def test_estimate_ocr_quality(self):
103        """
104        Match OCR annotations to TEI text and compute edit distance, WER, and CER.
105        """
106
107        def _text_from_range(root, ns, facs):
108            text = ""
109            start_pb = root.xpath(f".//tei:pb[contains(@facs, '{facs}')]", namespaces={"tei": ns["tei_ns"][1:-1]})
110            if len(start_pb) != 1 :
111                for e in root.find(f".//{ns['tei_ns']}body").iter():
112                    if e.text is not None:
113                        text += f" {' '.join([_.strip() for _ in e.text.splitlines() if _.strip() != ''])}"
114            else:
115                start_pb = start_pb[0]
116                start = False
117                for e in root.iter():
118                    if start and e.tag.endswith("}pb"):
119                        start = False
120                        break
121                    if e == start_pb:
122                        start = True
123                    if start:
124                        if e.text is not None:
125                            text += f" {' '.join([_.strip() for _ in e.text.splitlines() if _.strip() != ''])}"
126            return text
127
128        def _mk_string_list(l, text):
129            """
130            make a list of strings of len == len(annotation)
131            """
132            str_list = []
133            start = 0
134            while True:
135                str_list.append(text[start:start+l+1])
136                start += 1
137                if start + l + 1 == len(text):
138                    break
139            return str_list
140
141        def _get_most_probable_line(annotation, text):
142            most_probable_line = None
143            prob = None
144            l = len(annotation)
145
146            for start in range(0, len(text) - l):
147                s = text[start:start + l + 1]
148                lev = nltk.edit_distance(annotation.lower().strip(), s.lower().strip())
149
150                if prob is None or lev < prob:
151                    prob = lev
152                    most_probable_line = s
153                    if prob == 1 and annotation.endswith('-') and not s.endswith('-'):
154                        break
155            return most_probable_line, prob
156
157        rows = []
158        cols = [
159            "motion",
160            "parliament_year",
161            "chamber",
162            "annotation",
163            "most_probable_line",
164            "lev",
165            "wer",
166            "cer"
167        ]
168        for motion in tqdm([_ for _ in self.objective_reality["file"].unique() if "reg" not in _ and "fört" not in _]):
169            facs = motion.split("/")[-1].replace("-", "_").split('_')[-1]
170            if motion in self.file_mapping:
171                xml_file = self.file_mapping[motion]
172            else:
173                xml_file = f"data/{'/'.join(motion.split('riksdagen-motions-pdf/')[1].split('/')[:-1]).replace('_', '-')}.xml"
174            annotations = [_ for _ in self.objective_reality.loc[self.objective_reality["file"] == motion, "text"].tolist() if pd.notnull(_)]
175            py = self.objective_reality.loc[self.objective_reality["file"] == motion, "parliament_year"].unique().tolist()[0]
176            chamber = self.objective_reality.loc[self.objective_reality["file"] == motion, "chamber"].unique().tolist()[0]
177            root, ns = parse_tei(xml_file)
178            candidate_text = _text_from_range(root, ns, facs)
179
180            for annotation in annotations:
181                if annotation is None:
182                    continue
183                most_probable_line, lev = _get_most_probable_line(annotation, candidate_text)
184                if most_probable_line is None:
185                    self.match_errors.append([motion, xml_file, annotation])
186                    continue
187                wer = float(self.wer_fn(annotation, most_probable_line))
188                cer = lev/len(annotation)
189                rows.append([
190                    xml_file,
191                    py,
192                    chamber,
193                    annotation,
194                    most_probable_line,
195                    lev,
196                    wer,
197                    cer
198                ])
199        type(self).most_probable_lines = pd.DataFrame(rows, columns = cols)

103    def test_estimate_ocr_quality(self):
104        """
105        Match OCR annotations to TEI text and compute edit distance, WER, and CER.
106        """
107
108        def _text_from_range(root, ns, facs):
109            text = ""
110            start_pb = root.xpath(f".//tei:pb[contains(@facs, '{facs}')]", namespaces={"tei": ns["tei_ns"][1:-1]})
111            if len(start_pb) != 1 :
112                for e in root.find(f".//{ns['tei_ns']}body").iter():
113                    if e.text is not None:
114                        text += f" {' '.join([_.strip() for _ in e.text.splitlines() if _.strip() != ''])}"
115            else:
116                start_pb = start_pb[0]
117                start = False
118                for e in root.iter():
119                    if start and e.tag.endswith("}pb"):
120                        start = False
121                        break
122                    if e == start_pb:
123                        start = True
124                    if start:
125                        if e.text is not None:
126                            text += f" {' '.join([_.strip() for _ in e.text.splitlines() if _.strip() != ''])}"
127            return text
128
129        def _mk_string_list(l, text):
130            """
131            make a list of strings of len == len(annotation)
132            """
133            str_list = []
134            start = 0
135            while True:
136                str_list.append(text[start:start+l+1])
137                start += 1
138                if start + l + 1 == len(text):
139                    break
140            return str_list
141
142        def _get_most_probable_line(annotation, text):
143            most_probable_line = None
144            prob = None
145            l = len(annotation)
146
147            for start in range(0, len(text) - l):
148                s = text[start:start + l + 1]
149                lev = nltk.edit_distance(annotation.lower().strip(), s.lower().strip())
150
151                if prob is None or lev < prob:
152                    prob = lev
153                    most_probable_line = s
154                    if prob == 1 and annotation.endswith('-') and not s.endswith('-'):
155                        break
156            return most_probable_line, prob
